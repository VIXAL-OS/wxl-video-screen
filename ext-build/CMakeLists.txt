cmake_minimum_required(VERSION 3.20)
project(VideoCinemaExtension LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CORE "${CMAKE_CURRENT_LIST_DIR}/../../wxl-selected/sources/WarcraftXL--wxl-core/wxl-core-1.1")
file(GLOB SDK_SRC "${CORE}/src/game/*.cpp")
file(GLOB EXT_SRC "${CMAKE_CURRENT_LIST_DIR}/../extension/src/*.cpp")
add_library(wxl-video-screen SHARED ${EXT_SRC} ${SDK_SRC})
target_include_directories(wxl-video-screen PRIVATE "${CORE}/include" "${CORE}/src" "${CMAKE_CURRENT_LIST_DIR}/../extension/src")
target_compile_definitions(wxl-video-screen PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS WXL_EXTENSION)
target_compile_options(wxl-video-screen PRIVATE /utf-8)
target_link_libraries(wxl-video-screen PRIVATE d3d9 shell32 gdi32 user32)
