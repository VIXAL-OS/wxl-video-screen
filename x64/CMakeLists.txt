cmake_minimum_required(VERSION 3.20)
project(videocinema LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build for Windows x64")
endif()
set(WXL_SDK "" CACHE PATH "WarcraftXL SDK root, ABI 0.1.46")
add_library(videocinema SHARED native/Mod.cpp native/Screen.cpp native/FilmQuad.cpp native/FilmTransport.cpp)
target_compile_features(videocinema PRIVATE cxx_std_17)
target_include_directories(videocinema PRIVATE "${WXL_SDK}/include")
target_compile_definitions(videocinema PRIVATE WXL_BUILDING_MOD NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
set_property(TARGET videocinema PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(videocinema PRIVATE d3d9)
